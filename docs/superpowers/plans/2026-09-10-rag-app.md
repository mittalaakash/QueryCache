# RAG Application Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the LangGraph trip-planner demo in this repo with a local RAG application: a synthetic ~100-document corpus, an incremental ingestion pipeline into pgvector, retrieval + generation over Ollama, a query/document UI, and observability (Langfuse tracing, structured logging, Prometheus metrics).

**Architecture:** A new, isolated `pgvector/pgvector:pg17` Postgres container (port 5433) stores `documents` (path/title/content hash) and `chunks` (content + `vector(768)` embedding). `ingest.py` walks a `documents/` folder of markdown files, hashes each one, and only re-chunks/re-embeds files whose hash changed (skip unchanged, delete removed) — that hash comparison is the incremental-ingestion mechanism. `rag.py` embeds a question, does a pgvector cosine top-k lookup, and streams a cited answer from `llama3.2` over the FastAPI app's existing SSE pattern. `main.py` exposes the query UI, a document add/edit endpoint that triggers ingestion for just that one file, and a `/metrics` endpoint.

**Tech Stack:** FastAPI, `langchain-ollama` (`ChatOllama`, `OllamaEmbeddings`), `psycopg` v3 + `pgvector` (Python package) against `pgvector/pgvector:pg17`, `prometheus-client`, `langfuse`, `python-dotenv`.

**Spec:** `docs/superpowers/specs/2026-09-10-rag-app-design.md`

## Global Constraints

- Vector store is a **new, isolated** Postgres container on port `5433` — never the existing `postgres` container on `:5432` (belongs to another project, `llmgateway`, with real data).
- Embeddings model: `nomic-embed-text` via local Ollama (768 dimensions — must match the `chunks.embedding` column width exactly).
- Generation model: `llama3.2` via local Ollama (already pulled).
- Tracing: Langfuse only. `langgraph` and `langsmith` are dropped from `requirements.txt`.
- `.env` is gitignored already — real secrets go there, never in `.env.example` or committed files.
- No new dependencies beyond: `psycopg[binary]`, `pgvector`, `prometheus-client` (everything else already in `requirements.txt` or stdlib).

```python
"""Character-based text chunking for embedding. A fixed-size sliding window
with overlap is enough at this corpus size/shape — no need for a heavier
splitter library."""


def split_into_chunks(text: str, size: int = 800, overlap: int = 100) -> list[str]:
    text = text.strip()
    if not text:
        return []
    if len(text) <= size:
        return [text]

    chunks = []
    step = size - overlap
    start = 0
    while start < len(text):
        chunk = text[start:start + size]
        chunks.append(chunk)
        if start + size >= len(text):
            break
        start += step
    return chunks
```

```python
"""All Postgres/pgvector I/O for the RAG app: schema management, document
and chunk CRUD, and the cosine-similarity search used at query time."""

import os

import psycopg
from pgvector.psycopg import register_vector


def get_connection() -> psycopg.Connection:
    conn = psycopg.connect(os.environ["DATABASE_URL"], autocommit=True)
    conn.execute("CREATE EXTENSION IF NOT EXISTS vector")
    register_vector(conn)
    return conn


def init_schema(conn: psycopg.Connection) -> None:
    conn.execute("""
        CREATE TABLE IF NOT EXISTS documents (
            id SERIAL PRIMARY KEY,
            path TEXT UNIQUE NOT NULL,
            title TEXT NOT NULL,
            content_hash TEXT NOT NULL,
            updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
        )
    """)
    conn.execute("""
        CREATE TABLE IF NOT EXISTS chunks (
            id SERIAL PRIMARY KEY,
            document_id INTEGER NOT NULL REFERENCES documents(id) ON DELETE CASCADE,
            chunk_index INTEGER NOT NULL,
            content TEXT NOT NULL,
            embedding vector(768) NOT NULL
        )
    """)
    conn.execute("""
        CREATE INDEX IF NOT EXISTS chunks_embedding_idx
        ON chunks USING ivfflat (embedding vector_cosine_ops)
    """)


def get_document(conn: psycopg.Connection, path: str) -> dict | None:
    row = conn.execute(
        "SELECT id, content_hash FROM documents WHERE path = %s", (path,)
    ).fetchone()
    if row is None:
        return None
    return {"id": row[0], "content_hash": row[1]}


def upsert_document(conn: psycopg.Connection, path: str, title: str, content_hash: str) -> int:
    row = conn.execute(
        """
        INSERT INTO documents (path, title, content_hash)
        VALUES (%s, %s, %s)
        ON CONFLICT (path) DO UPDATE
            SET title = EXCLUDED.title,
                content_hash = EXCLUDED.content_hash,
                updated_at = now()
        RETURNING id
        """,
        (path, title, content_hash),
    ).fetchone()
    return row[0]


def replace_chunks(
    conn: psycopg.Connection,
    document_id: int,
    chunks: list[str],
    embeddings: list[list[float]],
) -> None:
    with conn.transaction():
        conn.execute("DELETE FROM chunks WHERE document_id = %s", (document_id,))
        with conn.cursor() as cur:
            cur.executemany(
                """
                INSERT INTO chunks (document_id, chunk_index, content, embedding)
                VALUES (%s, %s, %s, %s)
                """,
                [(document_id, i, c, e) for i, (c, e) in enumerate(zip(chunks, embeddings))],
            )


def delete_documents_not_in(conn: psycopg.Connection, existing_paths: set[str]) -> int:
    rows = conn.execute(
        "DELETE FROM documents WHERE path != ALL(%s) RETURNING id",
        (list(existing_paths),),
    ).fetchall()
    return len(rows)


def list_documents(conn: psycopg.Connection) -> list[dict]:
    rows = conn.execute(
        "SELECT title, path, updated_at FROM documents ORDER BY updated_at DESC"
    ).fetchall()
    return [{"title": r[0], "path": r[1], "updated_at": r[2].isoformat()} for r in rows]


def search_chunks(conn: psycopg.Connection, query_embedding: list[float], k: int = 4) -> list[dict]:
    rows = conn.execute(
        """
        SELECT d.title, d.path, c.content, c.embedding <=> %s AS distance
        FROM chunks c
        JOIN documents d ON d.id = c.document_id
        ORDER BY distance ASC
        LIMIT %s
        """,
        (query_embedding, k),
    ).fetchall()
    return [{"title": r[0], "path": r[1], "content": r[2], "distance": r[3]} for r in rows]
```

This module is a thin wrapper over real SQL against a real database — verified by manual integration checks below rather than mocked unit tests.

```python
"""Incremental ingestion: walk documents/*.md, hash each file, and only
re-chunk/re-embed files whose content changed since the last run. Files
deleted from disk are removed from the store too."""

import hashlib
import logging
from pathlib import Path
from typing import Callable, Literal

import db
from chunking import split_into_chunks

EmbedFn = Callable[[list[str]], list[list[float]]]
Action = Literal["insert", "update", "skip"]

logger = logging.getLogger("ingest")


def decide_action(existing_hash: str | None, new_hash: str) -> Action:
    if existing_hash is None:
        return "insert"
    if existing_hash == new_hash:
        return "skip"
    return "update"


def _extract_title(content: str, fallback: str) -> str:
    for line in content.splitlines():
        line = line.strip()
        if line.startswith("# "):
            return line[2:].strip()
    return fallback


def ingest_file(conn, embed_fn: EmbedFn, path: Path) -> Action:
    content = path.read_text()
    new_hash = hashlib.sha256(content.encode()).hexdigest()

    existing = db.get_document(conn, str(path))
    action = decide_action(existing["content_hash"] if existing else None, new_hash)
    if action == "skip":
        return action

    title = _extract_title(content, fallback=path.stem)
    chunks = split_into_chunks(content)
    chunk_embeddings = embed_fn(chunks)

    doc_id = db.upsert_document(conn, str(path), title, new_hash)
    db.replace_chunks(conn, doc_id, chunks, chunk_embeddings)
    return action


def sync_documents(conn, embed_fn: EmbedFn, documents_dir: Path) -> dict:
    paths = sorted(documents_dir.glob("*.md"))
    disk_paths = {str(p) for p in paths}

    counts = {"insert": 0, "update": 0, "skip": 0}
    for path in paths:
        action = ingest_file(conn, embed_fn, path)
        counts[action] += 1

    counts["delete"] = db.delete_documents_not_in(conn, disk_paths)
    return counts


if __name__ == "__main__":
    from dotenv import load_dotenv

    load_dotenv()
    logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(name)s: %(message)s")

    from embeddings import embedder

    conn = db.get_connection()
    db.init_schema(conn)
    result = sync_documents(conn, embedder.embed_documents, Path("documents"))
    logger.info("ingestion complete: %s", result)
```

```python
CHARACTERS = ["Mira", "Owen", "Priya", "Teodor", "Lin", "Amara"]

    "an abandoned observatory",
    "a river town after the flood",

def generate_revenue_report(rng: random.Random, index: int) -> tuple[str, str]:
    quarter = rng.choice(QUARTERS)
    year = rng.choice(YEARS)
    region = rng.choice(REGIONS)
    revenue = rng.randint(800_000, 9_500_000)
    growth = rng.uniform(-8.5, 32.0)
    product_lines = rng.sample(PRODUCTS, k=3)

    rows = []
    remaining = revenue
    for i, product in enumerate(product_lines):
        share = remaining // (len(product_lines) - i) if i < len(product_lines) - 1 else remaining
        rows.append((product, share))
        remaining -= share

    title = f"{quarter} {year} Revenue Report - {region} ({index})"
    lines = [
        f"# {title}",
        "",
        f"Total revenue for {region} in {quarter} {year} reached **${revenue:,}**, "
        f"a {growth:+.1f}% change from the prior quarter.",
        "",
        "| Product | Revenue |",
        "|---|---|",
    ]
    lines += [f"| {product} | ${share:,} |" for product, share in rows]
    lines += [
        "",
        f"Regional headcount stood at {rng.randint(40, 900)} employees, "
        f"with an average deal size of ${rng.randint(1_200, 85_000):,}.",

def generate_story(rng: random.Random, index: int) -> tuple[str, str]:

        f"with an answer.\n\n"
        f"By the {rng.randint(2, 12)}th night, {who} stopped asking why, and started asking "

def generate_faq(rng: random.Random, index: int) -> tuple[str, str]:

    price = rng.choice([9, 29, 49, 99, 199])

    title = f"FAQ: {feature.title()} ({index})"
    body = (

        f"**Q: What's the default limit for {feature}?**\n"

        f"**Q: Who do I contact for {feature} issues?**\n"
        f"A: Support responds within {rng.choice([1, 2, 4, 24])} hours on business days."
    )
    return title, body


def generate_meeting_notes(rng: random.Random, index: int) -> tuple[str, str]:
    topic = rng.choice(TOPICS)
    attendees = rng.sample(ATTENDEES, k=rng.randint(2, 4))
    action_items = rng.randint(1, 5)

    title = f"Meeting Notes: {topic.title()} ({index})"
    lines = [f"# {title}", "", f"Attendees: {', '.join(attendees)}", "", "## Discussion", ""]
    lines.append(
        f"The team reviewed {topic} and agreed budget impact was roughly "
        f"${rng.randint(5_000, 250_000):,}, affecting {rng.randint(1, 12)} teams."
    )
    lines += ["", "## Action items"]
    for _ in range(action_items):
        owner = rng.choice(attendees)
        lines.append(f"- [ ] {owner} to follow up within {rng.randint(1, 14)} days")
    return title, "\n".join(lines)


def generate_howto(rng: random.Random, index: int) -> tuple[str, str]:
    task = rng.choice(TASKS)
    steps = rng.randint(3, 6)

    title = f"How to {task} ({index})"
    lines = [f"# {title}", ""]
    for i in range(1, steps + 1):
        lines.append(
            f"{i}. {task.capitalize()} step {i}: run the command and confirm the "
            f"response code is {rng.choice([200, 201, 204])}."
        )
    lines += [
        "",
        f"Typical completion time is under {rng.randint(1, 30)} minutes; "
        f"rollback window is {rng.randint(5, 60)} minutes.",
    ]
    return title, "\n".join(lines)


GENERATORS = [
    generate_revenue_report,
    generate_story,
    generate_faq,
    generate_meeting_notes,
    generate_howto,
]


def generate_corpus(output_dir: Path, seed: int = 42, per_template: int = 20) -> int:
    rng = random.Random(seed)
    output_dir.mkdir(exist_ok=True)

    count = 0
    for generator in GENERATORS:
        prefix = generator.__name__.removeprefix("generate_")
        for i in range(per_template):
            title, body = generator(rng, i)
            slug = re.sub(r"[^a-z0-9_-]", "", title.lower().replace(" ", "_"))[:40]
            path = output_dir / f"{prefix}_{i:02d}_{slug}.md"
            path.write_text(body)
            count += 1
    return count


if __name__ == "__main__":
    total = generate_corpus(Path("documents"))
    print(f"Generated {total} documents in documents/")
```

- [ ] **Step 1: Write `rag.py`**

```python
"""Retrieval-augmented generation: embed the question, pull the top-k
nearest chunks from pgvector, build a cited-context prompt, and stream the
answer from the local Ollama chat model."""

from typing import AsyncIterator

from langchain_core.messages import HumanMessage
from langchain_ollama import ChatOllama

import db
from embeddings import embedder

_llm = ChatOllama(model="llama3.2", temperature=0.2)

PROMPT_TEMPLATE = """Answer the question using ONLY the context below. Cite \
sources by title in brackets, e.g. [Q3 2024 Revenue Report]. If the context \
doesn't contain the answer, say so plainly instead of guessing.

Context:
{context}

Question: {question}
Answer:"""


def build_prompt(question: str, chunks: list[dict]) -> str:
    context = "\n\n".join(f"[{c['title']}]\n{c['content']}" for c in chunks)
    return PROMPT_TEMPLATE.format(context=context, question=question)


async def stream_query(
    conn, question: str, callbacks: list | None = None, k: int = 4
) -> AsyncIterator[dict]:
    query_embedding = embedder.embed_query(question)
    chunks = db.search_chunks(conn, query_embedding, k=k)

    yield {"type": "sources", "data": [{"title": c["title"], "path": c["path"]} for c in chunks]}

    prompt = build_prompt(question, chunks)
    config = {"callbacks": callbacks} if callbacks else {}
    async for chunk in _llm.astream([HumanMessage(content=prompt)], config=config):
        if chunk.content:
            yield {"type": "token", "data": chunk.content}

    yield {"type": "done"}
```

```python
"""FastAPI RAG app: serves the query UI, answers questions over the ingested
document corpus via SSE, accepts document add/edit requests that trigger
incremental re-ingestion, and exposes Prometheus metrics."""

import json
import logging
import re
import time
import uuid
from pathlib import Path

from dotenv import load_dotenv
from fastapi import FastAPI, Request, Response
from fastapi.responses import FileResponse, StreamingResponse
from fastapi.staticfiles import StaticFiles
from langfuse.langchain import CallbackHandler as LangfuseCallbackHandler

import db
import ingest
import metrics
import rag
from embeddings import embedder

load_dotenv()
logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(name)s: %(message)s")
logger = logging.getLogger("rag_app")

DOCUMENTS_DIR = Path("documents")

app = FastAPI()
app.mount("/static", StaticFiles(directory="static"), name="static")

conn = db.get_connection()
db.init_schema(conn)
# No-ops (logs a warning, sends nothing) if LANGFUSE_PUBLIC_KEY/SECRET_KEY aren't set.
langfuse_handler = LangfuseCallbackHandler()


@app.get("/")
def index():
    return FileResponse("static/index.html")


def _sse(event: dict) -> str:
    return f"data: {json.dumps(event)}\n\n"


def _safe_filename(title: str) -> str:
    # Strip to a safe slug so a title can't be used for path traversal.
    slug = re.sub(r"[^a-z0-9_-]", "", title.strip().lower().replace(" ", "_"))
    return f"{slug or uuid.uuid4().hex}.md"


@app.post("/query")
async def query(req: Request):
    body = await req.json()
    question = body["question"]

    async def stream():
        start = time.perf_counter()
        chunk_count = 0
        async for event in rag.stream_query(conn, question, callbacks=[langfuse_handler]):
            if event["type"] == "sources":
                chunk_count = len(event["data"])
                metrics.RETRIEVED_CHUNKS.observe(chunk_count)
            yield _sse(event)
        metrics.QUERY_COUNT.inc()
        metrics.QUERY_LATENCY.observe(time.perf_counter() - start)
        logger.info("query answered question=%r retrieved_chunks=%d", question, chunk_count)

    return StreamingResponse(stream(), media_type="text/event-stream")


@app.get("/documents")
def list_documents():
    return db.list_documents(conn)


@app.post("/documents")
async def upsert_document(req: Request):
    body = await req.json()
    title, content = body["title"], body["content"]

    path = DOCUMENTS_DIR / _safe_filename(title)
    path.write_text(content)

    action = ingest.ingest_file(conn, embedder.embed_documents, path)
    metrics.INGESTED_DOCUMENTS.labels(action=action).inc()
    logger.info("document %s via UI path=%s", action, path)

    return {"action": action, "documents": db.list_documents(conn)}


@app.get("/metrics")
def get_metrics():
    body, content_type = metrics.render()
    return Response(content=body, media_type=content_type)
```
